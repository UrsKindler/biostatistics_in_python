def nipals_algorithm(X, n_comps=2, tol=1e-6, max_iter=500):
    """
    Calculate PCA using NIPALS algorithm,ignore NaNs
    """
    X = X.values.copy() # Als Numpy Array, Kopie damit Original nicht verändert wird
    n_samples, n_features = X.shape
    scores = np.zeros((n_samples, n_comps))
    loadings = np.zeros((n_features, n_comps))
    eigenvalues = np.zeros(n_comps)
    
    # Iteriere über jede Komponente (PC1, PC2, ...)
    for k in range(n_comps):
        
        # Wähle Startvektor (Spalte mit max Varianz)
        var_per_col = np.nanvar(X, axis=0)
        t = X[:, np.argmax(var_per_col)]
        
        # Iteration bis Konvergenz
        for iter_i in range(max_iter):
            t_old = t.copy()
            
            # 1. Berechne Loadings (p) - Ignoriere NaNs mit nansum
            # Projektion von X auf t
            numerator = np.nansum(X * t[:, None], axis=0)
            denominator = np.nansum(t[:, None]**2, axis=0)
            p = numerator / denominator
            p = p / np.linalg.norm(p) # Normalisieren
            
            # 2. Berechne Scores (t) - Ignoriere NaNs mit nansum
            # Projektion von X auf p
            numerator = np.nansum(X * p[None, :], axis=1)
            denominator = np.nansum(p[None, :]**2, axis=1)
            t = numerator / denominator
            
            # Check Konvergenz
            if np.linalg.norm(t - t_old) < tol:
                break
        
        # Speichern
        scores[:, k] = t
        loadings[:, k] = p
        
        # Eigenvalue approximieren (Varianz dieser Komponente)
        eigenvalues[k] = np.nanvar(t) * np.sum(p**2) 
        
        # Deflation: Erklärte Daten abziehen
        # X_new = X_old - t * p'
        X -= np.outer(t, p)
        
    return scores, loadings, eigenvalues

def perform_nipals_pca(data, max_components=10):
    
    print("Principal Component Analysis using Custom NIPALS")
    print("(Algorithm explicitly ignores NaNs during calculation)")
    
    n_components = min(max_components, min(data.shape))
    
    # Rufe unseren eigenen Algorithmus auf
    # data ist hier schon standardisiert (preprocess_data_nipals)
    scores, loadings, eigenvalues = nipals_algorithm(data, n_comps=n_components)
    
    # Varianz berechnen
    # Gesamte Varianz in standardisierten Daten = Anzahl der Features (da Var=1 pro Feature)
    total_variance = data.shape[1] 
    explained_variance = (eigenvalues / total_variance) * 100
    
    # Display variance explanation
    print(f"Principal components retained: {n_components}")
    print(f"Variance explained by PC1: {explained_variance[0]:.1f}%")
    print(f"Cumulative variance (PC1-3): {explained_variance[:3].sum():.1f}%")
    
    # Für Kompatibilität geben wir ein Dummy-Objekt zurück, falls später pca.components_ gebraucht wird
    class DummyPCA:
        def __init__(self):
            self.components_ = loadings.T
            self.explained_variance_ratio_ = explained_variance / 100
            
    pca_dummy = DummyPCA()
    
    return scores, explained_variance, pca_dummy
